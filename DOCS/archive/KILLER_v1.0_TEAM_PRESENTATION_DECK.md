kfn calculate_fibonacci(n: Int) -> Int
  if n <= 1
    return n
  a = 0
  b = 1
  i = 2
  while i <= n
    temp = a + b

vs Go:
```go
// GO: More boilerplate
func fibonacci(n int) int {
  if n <= 1 {
    return n
  }
  a, b := 0, 1
  for i := 2; i <= n; i++ {
    a, b = b, a+b
  }
  return b
}
```
